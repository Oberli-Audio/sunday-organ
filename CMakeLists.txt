cmake_minimum_required(VERSION 3.25)

# Sunday Organ - an Oberli Audio plugin, started from Oberli-Audio/plugin-template.
# Build commands: DEVELOPING.md. Rules this file follows: ship-bob versioning-and-provenance.md
# (one VERSION file, provenance header, pins, platform settings), sound-design-bob D.1.6 (the
# sanitizer builds) and the harness contract (scripts/dsp-kit/harness-contract.md).

# One VERSION file feeds the project, the plugin's version string, the About panel and the
# release; the git tag is v + VERSION.
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" PRODUCT_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION")

# The oldest macOS the plugin runs on (10.13; arm64 cannot go below 11.0). Must be set before
# project(): after it an empty cache entry wins and the binary needs the macOS it was built on.
if (NOT CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "10.13" CACHE STRING "Minimum macOS deployment version" FORCE)
endif()

project(SundayOrgan VERSION ${PRODUCT_VERSION} LANGUAGES C CXX)

# Provenance for the About panel and a friend's report, read from git at configure time and
# written to one generated header (a target-wide definition would rebuild every JUCE module
# on each reconfigure). file(CONFIGURE) rewrites only on change.
execute_process(COMMAND git rev-parse --short=12 HEAD
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
                OUTPUT_VARIABLE BUILD_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
execute_process(COMMAND git status --porcelain --untracked-files=no
                WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
                OUTPUT_VARIABLE PRODUCT_GIT_STATUS OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if (NOT BUILD_SHA)
    set(BUILD_SHA "unknown")
endif()
if (PRODUCT_GIT_STATUS)
    set(BUILD_DIRTY 1)
else()
    set(BUILD_DIRTY 0)
endif()
string(TIMESTAMP BUILD_DATE "%Y-%m-%dT%H:%M:%SZ" UTC)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/BuildInfo.h" CONTENT
"#pragma once
// Generated by CMakeLists.txt from VERSION and git at configure time - do not edit.
#define PRODUCT_VERSION \"@PRODUCT_VERSION@\"
#define BUILD_SHA       \"@BUILD_SHA@\"
#define BUILD_DATE      \"@BUILD_DATE@\"
#define BUILD_DIRTY     @BUILD_DIRTY@
" @ONLY)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if (APPLE AND NOT CMAKE_OSX_ARCHITECTURES)
    # Universal by default: Ableton Live 10.1 under Rosetta loads x86_64, a friend's Live 11 or
    # 12 on an M-series loads arm64. The sanitizer presets set arm64 only.
    set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "Build architectures for macOS" FORCE)
endif()

# Release symbols (ship-bob windows-packaging.md, macos-signing.md): dSYM / PDB per format,
# the shipped binary stripped on macOS. Set by the release preset and the workflow.
option(PRODUCT_RELEASE_SYMBOLS "Keep a dSYM / PDB per format and strip the shipped binary" OFF)
if (PRODUCT_RELEASE_SYMBOLS)
    if (MSVC)
        target_compile_options(SundayOrgan PUBLIC $<$<CONFIG:Release>:/Zi>)
        target_link_options(SundayOrgan PUBLIC $<$<CONFIG:Release>:/DEBUG /OPT:REF /OPT:ICF>)
    elseif (APPLE)
        target_compile_options(SundayOrgan PUBLIC $<$<CONFIG:Release>:-g>)
        foreach(format VST3 Standalone)
            if (TARGET SundayOrgan_${format})
                # with LTO the linker deletes its object (/tmp/lto.o) before dsymutil reads it: keep it
                target_link_options(SundayOrgan_${format} PRIVATE
                    "$<$<CONFIG:Release>:LINKER:-object_path_lto,${CMAKE_CURRENT_BINARY_DIR}/lto-${format}.o>")
                # strip runs after JUCE's ad-hoc signature and breaks it; macOS will not load a
                # broken signature (on Apple Silicon and under Rosetta), so sign ad hoc again last
                add_custom_command(TARGET SundayOrgan_${format} POST_BUILD
                    COMMAND dsymutil "$<TARGET_FILE:SundayOrgan_${format}>" -o "$<TARGET_BUNDLE_DIR:SundayOrgan_${format}>.dSYM"
                    COMMAND strip -x "$<TARGET_FILE:SundayOrgan_${format}>"
                    COMMAND codesign --force --sign - "$<TARGET_BUNDLE_DIR:SundayOrgan_${format}>"
                    COMMENT "dSYM, strip and ad-hoc signature: ${format}"
                    VERBATIM)
            endif()
        endforeach()
    endif()
endif()

# Memory and UB safety (D.1.6): PUBLIC so the runtime is linked into the loadable bundle and
# into the harness. MSVC has ASan only.
option(PRODUCT_ASAN "Build the plugin and the harness with AddressSanitizer + UndefinedBehaviorSanitizer" OFF)
if (PRODUCT_ASAN)
    if (MSVC)
        target_compile_options(SundayOrgan PUBLIC /fsanitize=address)
        target_link_options(SundayOrgan PUBLIC /INCREMENTAL:NO)
    else()
        target_compile_options(SundayOrgan PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(SundayOrgan PUBLIC -fsanitize=address,undefined)
    endif()
endif()
option(PRODUCT_TSAN "Build the plugin and the harness with ThreadSanitizer (Tier F)" OFF)
if (PRODUCT_TSAN AND NOT MSVC)
    target_compile_options(SundayOrgan PUBLIC -fsanitize=thread)
    target_link_options(SundayOrgan PUBLIC -fsanitize=thread)
endif()

# Real-time safety (rule 1): the harness with RealtimeSanitizer; needs LLVM 20+ (the rtsan
# preset takes the compiler from $LLVM_ROOT, set by `toolchain.py env`). processBlock runs as a
# real-time context and any allocation, lock or system call aborts with a stack trace.
option(PRODUCT_RTSAN "Build the harness with RealtimeSanitizer" OFF)

# ---------------------------------------------------------------------------
# Tests (G3 and G6): the suite drives the real DSP through the harness (tools/tests/test_plugin.py,
# from the kit's test_template.py), plus the table and bank lints. Labels: fast (every build,
# under 60 s), render (goldens), lint (the CSVs). Needs python3 with numpy and scipy; without
# them nothing is registered and a plain plugin build never fails.
#   ctest --test-dir build -L fast --output-on-failure
# ---------------------------------------------------------------------------
find_package(Python3 COMPONENTS Interpreter)
if (Python3_FOUND)
    execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import numpy, scipy" RESULT_VARIABLE PRODUCT_PY_DEPS OUTPUT_QUIET ERROR_QUIET)
    if (PRODUCT_PY_DEPS EQUAL 0)
        enable_testing()
        set(HARNESS_ENV "HARNESS=$<TARGET_FILE:SundayOrganHarness>" "DSPKIT=${CMAKE_CURRENT_SOURCE_DIR}/scripts/dsp-kit"
            "PRESET_ROOT=${CMAKE_CURRENT_BINARY_DIR}/test-presets" "FACTORY_PRESETS=${CMAKE_CURRENT_SOURCE_DIR}/presets")
        foreach(label fast render)
            add_test(NAME suite_${label}
                     COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/tests/test_plugin.py" --label ${label}
                     WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
            set_tests_properties(suite_${label} PROPERTIES ENVIRONMENT "${HARNESS_ENV}" LABELS ${label})
        endforeach()
        add_test(NAME harness_contract
                 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/dsp-kit/dspkit.py" smoke "$<TARGET_FILE:SundayOrganHarness>"
                 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
        set_tests_properties(harness_contract PROPERTIES ENVIRONMENT "${HARNESS_ENV}" LABELS fast)
        add_test(NAME parameter_table
                 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/tests/check_table.py"
                         --harness "$<TARGET_FILE:SundayOrganHarness>" --kind ${PRODUCT_KIND}
                 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
        set_tests_properties(parameter_table PROPERTIES LABELS "fast;lint")
        add_test(NAME suite_features
                 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/tests/test_features.py"
                 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
        set_tests_properties(suite_features PROPERTIES ENVIRONMENT "${HARNESS_ENV};LESLIE_PROBE=$<TARGET_FILE:SundayOrganDopplerCheck>" LABELS render TIMEOUT 600)
        add_test(NAME doppler_moving_source
                 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/g5/doppler_check.py" "$<TARGET_FILE:SundayOrganDopplerCheck>"
                 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
        set_tests_properties(doppler_moving_source PROPERTIES ENVIRONMENT "${HARNESS_ENV}" LABELS "fast;render")
        add_test(NAME design_sources_in_git
                 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/tests/check_sources.py" "${CMAKE_CURRENT_SOURCE_DIR}")
        set_tests_properties(design_sources_in_git PROPERTIES LABELS "fast;lint")
        add_test(NAME preset_bank
                 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/dsp-kit/presetlint.py" bank
                         "${CMAKE_CURRENT_SOURCE_DIR}/presets.csv" --tier P --track ${PRODUCT_KIND}
                         --categories Jazz,Gospel,Worship,Rock,Ballad,Split --harness "$<TARGET_FILE:SundayOrganHarness>"
                 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
        set_tests_properties(preset_bank PROPERTIES ENVIRONMENT "${HARNESS_ENV}" LABELS "fast;lint")
    else()
        message(STATUS "python3 without numpy and scipy: the test suite is not registered")
    endif()
endif()
